### Module 1 — Authentication & Session Management

```python
# chartink/auth.py
"""
Session handler: login → maintain cookies → auto-refresh on 401/403.
Chartink uses CSRF token + session cookie pair.
"""

import os
import time
import requests
from loguru import logger
from dotenv import load_dotenv
from typing import Optional

load_dotenv()

CHARTINK_BASE    = "https://chartink.com"
LOGIN_URL        = f"{CHARTINK_BASE}/login"
SCREENER_PROCESS = f"{CHARTINK_BASE}/screener/process"

class ChartinkSession:
    def __init__(self):
        self.session   = requests.Session()
        self._csrf     = None
        self._logged_in = False
        self._setup_headers()

    def _setup_headers(self):
        self.session.headers.update({
            "User-Agent": (
                "Mozilla/5.0 (Windows NT 10.0; Win64; x64) "
                "AppleWebKit/537.36 (KHTML, like Gecko) "
                "Chrome/126.0.0.0 Safari/537.36"
            ),
            "Accept":          "application/json, text/javascript, */*; q=0.01",
            "Accept-Language": "en-US,en;q=0.9",
            "X-Requested-With": "XMLHttpRequest",
            "Referer":         CHARTINK_BASE,
        })

    def _fetch_csrf(self) -> str:
        """Extract Laravel CSRF token from login page."""
        r = self.session.get(LOGIN_URL, timeout=15)
        r.raise_for_status()
        # Token lives in <meta name="csrf-token" content="...">
        import re
        match = re.search(r'<meta name="csrf-token" content="([^"]+)"', r.text)
        if not match:
            raise RuntimeError("CSRF token not found — Chartink may have changed login page structure")
        return match.group(1)

    def login(self) -> bool:
        """
        Authenticate with Chartink using credentials from .env:
            CHARTINK_EMAIL=your@email.com
            CHARTINK_PASSWORD=yourpassword
        """
        email    = os.environ["CHARTINK_EMAIL"]
        password = os.environ["CHARTINK_PASSWORD"]

        self._csrf = self._fetch_csrf()

        payload = {
            "_token":   self._csrf,
            "email":    email,
            "password": password,
        }
        r = self.session.post(
            f"{CHARTINK_BASE}/login",
            data=payload,
            allow_redirects=True,
            timeout=15
        )

        # Successful login redirects to dashboard; session cookie set
        if r.status_code == 200 and "chartink_session" in self.session.cookies:
            self._logged_in = True
            # Update CSRF for subsequent API calls (may rotate post-login)
            import re
            match = re.search(r'<meta name="csrf-token" content="([^"]+)"', r.text)
            if match:
                self._csrf = match.group(1)
            logger.info("Chartink login successful")
            return True

        logger.error(f"Login failed: HTTP {r.status_code}")
        return False

    def fetch_screener(
        self,
        scan_clause: str,
        retries: int = 3,
        backoff: float = 2.0
    ) -> dict:
        """
        POST to /screener/process with a scan clause.

        Args:
            scan_clause: The screener condition string copied from DevTools payload.
                         Example: "( {cash} ( weekly rsi( 14 ) < 30 ) )"

        Returns:
            Parsed JSON dict: {"data": [...], "count": N}
        """
        if not self._logged_in:
            raise RuntimeError("Call .login() before fetching screeners")

        headers = {
            "X-CSRF-TOKEN": self._csrf,
            "Content-Type": "application/x-www-form-urlencoded; charset=UTF-8",
        }

        for attempt in range(1, retries + 1):
            try:
                r = self.session.post(
                    SCREENER_PROCESS,
                    data={"scan_clause": scan_clause},
                    headers=headers,
                    timeout=30
                )
                if r.status_code == 419:  # CSRF expired
                    logger.warning("CSRF expired, refreshing token")
                    self._csrf = self._fetch_csrf()
                    headers["X-CSRF-TOKEN"] = self._csrf
                    continue
                if r.status_code in (401, 403):
                    logger.warning("Session expired, re-logging in")
                    self.login()
                    continue
                r.raise_for_status()
                return r.json()

            except requests.exceptions.Timeout:
                logger.warning(f"Timeout on attempt {attempt}/{retries}")
                if attempt < retries:
                    time.sleep(backoff * attempt)
            except requests.exceptions.RequestException as e:
                logger.error(f"Request error: {e}")
                if attempt < retries:
                    time.sleep(backoff * attempt)

        raise RuntimeError(f"All {retries} attempts failed for scan_clause")

    def close(self):
        self.session.close()
```

```python
# chartink/extractor.py
"""
Fetches all screener widgets, normalizes JSON → DataFrame, saves to CSV/parquet.
"""

import time
import pandas as pd
from pathlib import Path
from datetime import datetime
from loguru import logger

from .auth import ChartinkSession
from .scan_clauses import ALL_DASHBOARDS

OUTPUT_DIR = Path("data/chartink")
OUTPUT_DIR.mkdir(parents=True, exist_ok=True)

# Minimum seconds between successive API calls (rate limiting)
REQUEST_DELAY_SECONDS = 2.5


def parse_screener_response(raw: dict, widget_name: str) -> pd.DataFrame:
    """
    Chartink screener JSON structure:
    {
        "data": [
            {
                "nsecode": "RELIANCE",
                "bsecode": "500325",
                "company_name": "Reliance Industries",
                "close": 2950.5,
                "per_chg": 1.23,
                ...
            },
            ...
        ],
        "count": 150
    }
    Column names vary by screener; all numeric fields come as strings.
    """
    records = raw.get("data", [])
    if not records:
        logger.warning(f"[{widget_name}] Empty data returned (count={raw.get('count', 0)})")
        return pd.DataFrame()

    df = pd.DataFrame(records)
    df["_widget"]    = widget_name
    df["_fetched_at"] = datetime.now().isoformat(timespec="seconds")

    # Coerce numeric columns
    skip_cols = {"nsecode", "bsecode", "company_name", "_widget", "_fetched_at"}
    for col in df.columns:
        if col not in skip_cols:
            df[col] = pd.to_numeric(df[col], errors="ignore")

    return df


def fetch_dashboard(
    session: ChartinkSession,
    dashboard_name: str,
    widgets: dict[str, str],
    save: bool = True,
) -> dict[str, pd.DataFrame]:
    """
    Fetch all widgets in a dashboard. Returns {widget_name: DataFrame}.
    """
    results: dict[str, pd.DataFrame] = {}
    timestamp = datetime.now().strftime("%Y%m%d_%H%M")

    for idx, (widget_name, scan_clause) in enumerate(widgets.items()):
        logger.info(f"[{dashboard_name}] Fetching widget: {widget_name} ({idx+1}/{len(widgets)})")

        try:
            raw = session.fetch_screener(scan_clause)
            df  = parse_screener_response(raw, widget_name)
            results[widget_name] = df

            if save and not df.empty:
                out_path = OUTPUT_DIR / dashboard_name / f"{widget_name}_{timestamp}.csv"
                out_path.parent.mkdir(parents=True, exist_ok=True)
                df.to_csv(out_path, index=False)
                logger.info(f"  → Saved {len(df)} rows to {out_path}")

        except Exception as e:
            logger.error(f"  → FAILED [{widget_name}]: {e}")
            results[widget_name] = pd.DataFrame()

        # Rate limiting: don't hammer Chartink
        if idx < len(widgets) - 1:
            time.sleep(REQUEST_DELAY_SECONDS)

    return results


def fetch_all_dashboards(save: bool = True) -> dict[str, dict[str, pd.DataFrame]]:
    """
    Entry point: authenticates, fetches all configured dashboards.
    """
    session = ChartinkSession()
    all_results = {}

    try:
        if not session.login():
            raise RuntimeError("Authentication failed — check credentials in .env")

        for dashboard_name, widgets in ALL_DASHBOARDS.items():
            logger.info(f"━━━ Dashboard: {dashboard_name} ({len(widgets)} widgets) ━━━")
            all_results[dashboard_name] = fetch_dashboard(
                session, dashboard_name, widgets, save=save
            )
            time.sleep(REQUEST_DELAY_SECONDS * 2)  # Extra pause between dashboards

    finally:
        session.close()

    return all_results


def merge_dashboard_to_wide(
    results: dict[str, pd.DataFrame],
    join_col: str = "nsecode"
) -> pd.DataFrame:
    """
    Pivot multi-widget fetch into wide format: one row per stock,
    one column set per widget. Useful for the RSI dashboard where
    you want all timeframes side-by-side.
    """
    frames = []
    for widget_name, df in results.items():
        if df.empty:
            continue
        # Prefix non-key columns with widget name
        rename = {
            c: f"{widget_name}__{c}"
            for c in df.columns
            if c not in {join_col, "company_name", "bsecode", "_fetched_at", "_widget"}
        }
        frames.append(df.rename(columns=rename))

    if not frames:
        return pd.DataFrame()

    wide = frames[0]
    for f in frames[1:]:
        common_cols = [join_col, "company_name"]
        merge_on = [c for c in common_cols if c in f.columns]
        wide = wide.merge(
            f[[c for c in f.columns if c not in {"_widget", "bsecode"}]],
            on=merge_on,
            how="outer",
            suffixes=("", "_dup")
        )
        # Drop accidental duplicates
        wide = wide[[c for c in wide.columns if not c.endswith("_dup")]]

    return wide
```

```python
# chartink/playwright_downloader.py
"""
Playwright fallback: navigate to dashboard, trigger CSV download per widget.
Slower but handles edge cases where scan_clause is obfuscated or POST fails.

Install: pip install playwright && playwright install chromium
"""

import asyncio
import os
from pathlib import Path
from datetime import datetime
from loguru import logger
from playwright.async_api import async_playwright, Page, Download

DOWNLOAD_DIR = Path("data/chartink/downloads")
DOWNLOAD_DIR.mkdir(parents=True, exist_ok=True)

CHARTINK_BASE = "https://chartink.com"


async def _login(page: Page) -> None:
    await page.goto(f"{CHARTINK_BASE}/login", wait_until="networkidle")
    await page.fill("input[name='email']",    os.environ["CHARTINK_EMAIL"])
    await page.fill("input[name='password']", os.environ["CHARTINK_PASSWORD"])
    async with page.expect_navigation(wait_until="networkidle"):
        await page.click("button[type='submit']")
    logger.info("Playwright login complete")


async def _intercept_screener_api(
    page: Page,
    dashboard_url: str
) -> list[dict]:
    """
    Navigate to dashboard; intercept all /screener/process POST responses.
    Returns list of raw JSON payloads (one per widget).
    """
    intercepted = []

    async def handle_response(response):
        if "/screener/process" in response.url and response.request.method == "POST":
            try:
                body = await response.json()
                req_body = response.request.post_data or ""
                intercepted.append({
                    "scan_clause": req_body,
                    "data":        body,
                })
                logger.debug(f"Intercepted screener: {len(body.get('data', []))} rows")
            except Exception as e:
                logger.warning(f"Failed to parse intercepted response: {e}")

    page.on("response", handle_response)
    await page.goto(dashboard_url, wait_until="networkidle", timeout=60_000)
    # Wait for all widgets to load (adjust based on widget count)
    await page.wait_for_timeout(5_000)

    return intercepted


async def download_all_widgets_playwright(dashboard_url: str) -> list[dict]:
    """
    Open dashboard via Playwright, intercept API calls, return raw payloads.
    Each payload can be passed to parse_screener_response().
    """
    async with async_playwright() as pw:
        browser = await pw.chromium.launch(
            headless=True,
            args=["--no-sandbox", "--disable-dev-shm-usage"]
        )
        context = await browser.new_context(
            viewport={"width": 1920, "height": 1080},
            accept_downloads=True
        )
        page = await context.new_page()

        await _login(page)
        payloads = await _intercept_screener_api(page, dashboard_url)

        await browser.close()
        return payloads


# Sync wrapper for use in non-async pipelines
def fetch_via_playwright(dashboard_url: str) -> list[dict]:
    return asyncio.run(download_all_widgets_playwright(dashboard_url))
```

```python
        return

    # Default: fetch everything
    results = fetch_all_dashboards(save=not args.no_save)
    for dashboard, widgets in results.items():
        total = sum(len(df) for df in widgets.values())
        logger.info(f"{dashboard}: {total} total rows across {len(widgets)} widgets")


if __name__ == "__main__":
    main()
```
